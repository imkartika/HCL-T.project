#include #include #include using namespace std; int main() { string password; bool hasUpper = false; bool hasLower = false; bool hasDigit = false; bool hasSpecial = false; cout << "================================\n"; cout << " PASSWORD SECURITY TOOL\n"; cout << "================================\n"; cout << "Enter a test password: "; getline(cin, password); for (char ch : password) { if (isupper(static_cast(ch))) { hasUpper = true; } else if (islower(static_cast(ch))) { hasLower = true; } else if (isdigit(static_cast(ch))) { hasDigit = true; } else { hasSpecial = true; } } int score = 0; if (password.length() >= 12) score++; if (hasUpper) score++; if (hasLower) score++; if (hasDigit) score++; if (hasSpecial) score++; cout << "\n----- PASSWORD ANALYSIS -----\n"; cout << "Length: " << password.length() << endl; cout << "Uppercase: " << (hasUpper ? "Yes" : "No") << endl; cout << "Lowercase: " << (hasLower ? "Yes" : "No") << endl; cout << "Numbers: " << (hasDigit ? "Yes" : "No") << endl; cout << "Special characters: " << (hasSpecial ? "Yes" : "No") << endl; cout << "\nStrength: "; if (password.length() < 8 || score <= 2) cout << "Weak"; else if (password.length() >= 12 && score >= 4) cout << "Strong"; else cout << "Moderate"; cout << endl; return 0; }
